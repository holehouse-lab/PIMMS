# Documentation build requirements (used by Read the Docs and for local builds).
#
# The pimms package itself is NOT installed for the docs build: its compiled Cython
# extensions and heavy runtime dependencies (mdtraj, scipy, dateutil) are mocked in
# conf.py, so nothing needs to be compiled. numpy is a real dependency because
# pimms/CONFIG.py (the single source of truth for the keyword reference) imports it
# at module load time.
sphinx>=7,<9
sphinx_rtd_theme>=2.0
numpy

# versioningit lets conf.py read the version straight from the git tags, so the version
# and release date shown on the index page are correct even though PIMMS itself is not
# installed here. Without it conf.py has nothing left to fall back to in this
# environment and the page reports "unknown". (See also the post_checkout job in
# .readthedocs.yaml, which un-shallows the clone so a tag is actually reachable.)
versioningit>=2
